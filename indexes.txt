    <script>
        (function() {
            const basePath = (() => {
                const path = window.location.pathname;
                if (path.includes('/about/') || path.includes('/redes/') || path.includes('/equipo/')) {
                    return '../';
                }
                return '';
            })();

            function getValidImgSrc(src) {
                if (!src) return '';
                if (src.startsWith('http://') || src.startsWith('https://')) return src;
                return basePath + src;
            }

            function getBadgeHTML(tipo) {
                if (tipo === 'Entrevista') {
                    return '<span class="badge-interview">ENTREVISTA</span>';
                }
                return '<span class="badge-default">' + (tipo || 'PUBLICACIÓN').toUpperCase() + '</span>';
            }

            function escapeHtml(str) {
                if (!str) return '';
                return str.replace(/[&<>]/g, function(m) {
                    if (m === '&') return '&amp;';
                    if (m === '<') return '&lt;';
                    if (m === '>') return '&gt;';
                    return m;
                });
            }

            async function loadGallery() {
                const container = document.getElementById('gallery-container');
                if (!container) return;
                try {
                    const res = await fetch(basePath + 'articulos/sitios.json?t=' + Date.now());
                    if (!res.ok) return;
                    const data = await res.json();
                    data.sort((a, b) => parseInt(a.posicion, 10) - parseInt(b.posicion, 10));
                    container.innerHTML = data.map(item => {
                        const imgSrc = getValidImgSrc(item.imagen);
                        return `
                            <div class="gallery-card">
                                <div class="gallery-img-container">
                                    <img src="${imgSrc}" onerror="this.outerHTML='<div class=\\'no-media\\'>NO MEDIA</div>'">
                                    <a href="${item.sitio}" target="_blank" class="gallery-btn">
                                        Ver sitio <span class="icon-arrow-top-right"></span>
                                    </a>
                                </div>
                                <div class="gallery-content">
                                    <div class="gallery-site-name">${escapeHtml(item.nombre_sitio)}</div>
                                    <h3 class="gallery-card-title">${escapeHtml(item.titulo)}</h3>
                                    <div class="gallery-badge-container">${getBadgeHTML(item.tipo)}</div>
                                </div>
                            </div>
                        `;
                    }).join('');
                } catch(e) {
                    console.error('Error loading gallery:', e);
                }
            }

            if (document.readyState === 'loading') {
                document.addEventListener('DOMContentLoaded', loadGallery);
            } else {
                loadGallery();
            }
        })();
    </script>
